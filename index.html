Visão Geral do Projeto Avançado

Este documento apresenta a versão ultra completa do portal Ajuda Idoso, unificando HTML, CSS avançado e JavaScript interativo em um único arquivo de alta performance para publicação no GitHub Pages.

Tema Claro / Escuro / Alto Contraste Leitor de Voz Web Speech API Lembretes Interativos de Remédios Botão SOS WhatsApp Design Ultrachamativo

1. Novas Funcionalidades Incorporadas

  • Alternador de Temas: Suporte dinâmico aos temas Claro (Suave), Escuro (Dark Mode) e Alto Contraste Acessível.
  • Cores Vibração Máxima: Cores estrategicamente vibrantes (Amarelo Ouro, Vermelho Alerta, Azul Vivo e Verde Esperança) para dar visibilidade imediata às seções cruciais.
  • Lembrete Interativo de Remédios: Permite adicionar o nome e o horário do medicamento para exibição na tela.
  • Sintetizador de Voz (Ler em Voz Alta): Botão que lê os avisos principais do site em português para idosos com baixa visão.
  • Rede de Emergência Direta: Chamada rápida para SAMU, Bombeiros, Polícia e Botão Direto para o WhatsApp de um familiar cadastrado.
  • Atividades Cognitivas & Exercícios de Memória: Caixas coloridas e destacadas com jogos simples e dicas de saúde.

    <script>
        let scale = 1.1;

        function changeFontSize(delta) {
            scale += delta;
            if (scale < 0.9) scale = 0.9;
            if (scale > 1.6) scale = 1.6;
            document.documentElement.style.setProperty('--font-scale', scale + 'rem');
        }

        function toggleTheme(theme) {
            document.body.classList.remove('dark-mode', 'high-contrast');
            if (theme === 'dark') document.body.classList.add('dark-mode');
            if (theme === 'contrast') document.body.classList.add('high-contrast');
        }

        function addRemedy() {
            const name = document.getElementById('remedyName').value;
            const time = document.getElementById('remedyTime').value;
            if (name && time) {
                const li = document.createElement('li');
                li.className = 'remedy-item';
                li.innerHTML = `💊 ${name} - ${time}`;
                document.getElementById('remedyList').appendChild(li);
                document.getElementById('remedyName').value = '';
                document.getElementById('remedyTime').value = '';
            }
        }

        function checkAnswer(val) {
            const feedback = document.getElementById('gameFeedback');
            if (val === 30) {
                feedback.innerText = "🎉 Parabéns! Você acertou!";
                feedback.style.color = "#16a34a";
            } else {
                feedback.innerText = "❌ Tente novamente!";
                feedback.style.color = "#dc2626";
            }
        }

        function readPage() {
            const text = "Bem-vindo ao Ajuda Idoso. Em caso de emergência, utilize os botões vermelhos do topo da página.";
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = 'pt-BR';
            window.speechSynthesis.speak(utterance);
        }

        function submitForm(e) {
            e.preventDefault();
            const nome = document.getElementById('nome').value;
            document.getElementById('statusMsg').innerText = `Obrigado ${nome}! Seu pedido foi registrado com sucesso.`;
            document.getElementById('contactForm').reset();
        }
    </script>
